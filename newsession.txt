import * as crypto from 'crypto';

/**
 * Generates the PayWay HMAC signature.
 * Concatenates the values of the fields in the order specified by the fieldList,
 * substituting empty string for undefined/null values.
 */
export function generateHmac(
  payload: Record<string, any>,
  fieldList: string[],
  apiKey: string
): string {
  const concatenated = fieldList
    .map(field => {
      const val = payload[field];
      if (val === undefined || val === null) {
        return '';
      }
      return String(val);
    })
    .join('');

  return crypto
    .createHmac('sha512', apiKey)
    .update(concatenated)
    .digest('base64');
}

/**
 * Encrypts data using RSA public key encryption with PKCS1 padding.
 * Plaintext is split into 117-byte chunks, each chunk is encrypted,
 * the encrypted chunks are concatenated, and the final buffer is Base64 encoded.
 */
export function encryptMerchantAuth(
  data: Record<string, any>,
  publicKeyPem: string
): string {
  const jsonStr = JSON.stringify(data);
  const buffer = Buffer.from(jsonStr, 'utf8');
  const chunkSize = 117;
  const encryptedChunks: Buffer[] = [];

  for (let i = 0; i < buffer.length; i += chunkSize) {
    const chunk = buffer.subarray(i, i + chunkSize);
    const encrypted = crypto.publicEncrypt(
      {
        key: publicKeyPem,
        padding: crypto.constants.RSA_PKCS1_PADDING,
      },
      chunk
    );
    encryptedChunks.push(encrypted);
  }

  return Buffer.concat(encryptedChunks).toString('base64');
}

/**
 * Verifies a webhook signature using PayWay's sorted-key signature validation algorithm.
 */
export function verifyCallbackSignature(
  body: Record<string, any>,
  receivedSignature: string,
  apiKey: string
): boolean {
  // Sort response keys ascending
  const sortedKeys = Object.keys(body).sort();

  // Concatenate all values, JSON-encoding any array/object values
  const concatenated = sortedKeys
    .map(key => {
      const val = body[key];
      if (val === undefined || val === null) {
        return '';
      }
      if (typeof val === 'object') {
        return JSON.stringify(val);
      }
      return String(val);
    })
    .join('');

  const computedSignature = crypto
    .createHmac('sha512', apiKey)
    .update(concatenated)
    .digest('base64');

  const computedBuf = Buffer.from(computedSignature);
  const receivedBuf = Buffer.from(receivedSignature);

  if (computedBuf.length !== receivedBuf.length) {
    return false;
  }

  return crypto.timingSafeEqual(computedBuf, receivedBuf);
}

       and produces a hash mismatch (`code: 1`, "Wrong hash") with no
       further diagnostic detail.
    2. `req_time` MUST be UTC, formatted `YYYYMMDDHHmmss`, and generated
       at the moment of signing — not cached — since it is itself part of
       the signed payload.
    3. The callback (return_url) verification algorithm is DIFFERENT from
       request signing: sort response keys ascending, JSON-encode any
       array/object values, concatenate all values, HMAC-SHA512 against
       the same api_key, compare to the `X-PAYWAY-HMAC-SHA512` header
       using constant-time comparison. See the `PaymentCallback` webhook
       below. Do not reuse the request-signing function for this.
    4. `purchase` accepts `multipart/form-data`; `check-transaction` and
       `close-transaction` accept `application/json`. Do not assume a
       uniform content type across operations when generating client
       method signatures.
    5. `purchase`'s 200 response is polymorphic: HTML (hosted checkout
       page) by default, or a JSON object containing `qr_string`,
       `abapay_deeplink`, `checkout_qr_url` when
       `payment_option=abapay_khqr_deeplink`. Client method return types
       must model both branches — do not assume HTML-only.
    6. UNVERIFIED — confirm against sandbox before relying on it: the
       `check-transaction` and `get-transaction-list` paths below both
       include a literal `-2` suffix (`check-transaction-2`,
       `transaction-list-2`) as scraped directly from PayWay's docs
       export. This looks like it could be a vendor-side docs-platform
       artifact rather than the real production path. Treat as
       provisional pending a live sandbox call.
    7. `refund` uses a THIRD, distinct auth layer on top of HMAC: the
       `merchant_auth` field is an RSA-public-key-encrypted, then
       Base64-encoded, JSON blob of `{mc_id, tran_id, refund_amount}`.
       PayWay's reference implementation encrypts in 117-byte chunks
       (consistent with a 1024-bit RSA key under PKCS1 padding —
       worth flagging to ABA Bank as low by modern standards and
       confirming no larger key is available). This means your SDK's
       credential config needs to hold BOTH the HMAC `api_key` AND a
       separate RSA public key, and refund needs its own signing
       pipeline entirely distinct from every other operation. Do not
       try to unify it with `x-hmac-fields`.
    8. `refund` also uses `request_time` as its field name where every
       other endpoint uses `req_time` — this is very likely a vendor
       documentation inconsistency, not an intentional API difference,
       but the generated SDK method must honor it literally (send
       `request_time` on the wire for refund) or the request fails
       validation before the hash is even checked. Flag this loudly in
       code review; it's the kind of thing that gets "corrected" by
       someone skimming and then silently breaks in production.
    9. Path prefixes are not uniform: most operations live under
       `/api/payment-gateway/v1/...`, but `refund` lives under
       `/api/merchant-portal/merchant-access/online-transaction/refund`
       and `exchange-rate` lives directly under
       `/api/payment-gateway/v1/exchange-rate` (no `/payments/`
       segment). Do not assume a shared base path when generating
       client method routing.
    10. UNVERIFIED: `get-transaction-list`'s reference PHP hash sample
        concatenates `$to_date + $from_amount` using PHP's arithmetic
        `+` operator rather than `.` (string concatenation) as every
        other sample uses. This is almost certainly a copy-paste typo
        in PayWay's own docs (PHP would coerce both to numbers,
        producing garbage for date strings), but it's PayWay's
        reference implementation, so treat the correct on-wire behavior
        as unconfirmed until validated against a live sandbox call —
        do not silently "fix" it in your signing code without testing
        both interpretations.

type Verdict = 'PATH_OK' | 'PATH_MISSING' | 'NETWORK_ERR' | 'HMAC_REJECTED';

interface ProbeResult {
  path: string;
  status: number | 'ERR';
  statusText: string;
  bodySnippet: string;
  verdict: Verdict;
}

async function probe(
  path: string,
  body: Record<string, any>,
  hmacFields: string[]
): Promise<ProbeResult> {
  const payload = { ...body, merchant_id: MERCHANT_ID };
  payload.hash  = hmacSign(payload, hmacFields, API_KEY);

  let status: number | 'ERR' = 'ERR';
  let statusText = '';
  let rawBody    = '';

  try {
    const res = await fetch(`${BASE_URL}${path}`, {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify(payload),
    });
    status     = res.status;
    statusText = res.statusText;
    rawBody    = await res.text();
  } catch (err: any) {
    return {
      path,
      status:      'ERR',
      statusText:  err?.message ?? 'Network error',
      bodySnippet: '',
      verdict:     'NETWORK_ERR',
    };
  }

  const bodySnippet = rawBody.length > 350 ? rawBody.slice(0, 350) + '…' : rawBody;

  let verdict: Verdict;
  if (status === 200) {
    let json: any = null;
    try { json = JSON.parse(rawBody); } catch { /* HTML or plain text */ }

    if (json) {
      // code:"1" == "Wrong hash" → path reached, signing was off
      verdict = (json.status === 'FAILED' && String(json.code) === '1')
        ? 'HMAC_REJECTED'
        : 'PATH_OK';
    } else if (rawBody.trimStart().startsWith('<')) {
      // HTML error page → path probably doesn't exist
      verdict = 'PATH_MISSING';
    } else {
      verdict = 'PATH_OK';
    }
  } else if (status === 404) {
    verdict = 'PATH_MISSING';
  } else if (typeof status === 'number' && status >= 400) {
    // Other 4xx/5xx — server responded, so path exists
    verdict = 'PATH_OK';
  } else {
    verdict = 'PATH_MISSING';
  }

  return { path, status, statusText, bodySnippet, verdict };
}
