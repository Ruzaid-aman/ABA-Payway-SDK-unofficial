    Type Declaration

      • (
            transactionId: string,
            amount: number,
            currency?: "USD" | "KHR",
        ): Promise<
            {
                currency?: string;
                grand_total?: number;
                status: { code: string; message: string };
                total_refunded?: number;
                transaction_status?: string;
            },
        >

      • Parameters

        • transactionId: string

          The original purchase transaction ID.

        • amount: number

          The refund amount. Must be ≥ 0.01 for USD, ≥ 1 for KHR.

        • Optionalcurrency: "USD" | "KHR"

          The original transaction currency. Defaults to 'USD'. Providing the correct currency enables client-side minimum-amount validation (PayWay returns PTL04 if refund_amount < 0.01 USD).

        Returns Promise<
            {
                currency?: string;
                grand_total?: number;
                status: { code: string; message: string };
                total_refunded?: number;
                transaction_status?: string;
